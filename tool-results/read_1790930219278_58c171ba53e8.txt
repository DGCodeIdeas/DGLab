   320→    // --- Bootstrapper Circuit Breaker Tests (P6 closure, doctrine §4.5.3) ---
   321→
   322→    /**
   323→     * Per doctrine §4.5.3: each BootstrapperInterface::bootstrap() call MUST
   324→     * be wrapped in a per-bootstrapper wall-clock budget of 5 seconds.
   325→     * Exceeding the budget throws BootstrapperTimeoutExceeded (KernelException,
   326→     * class Permanent-Local per doctrine §2 taxonomy). The Kernel transitions
   327→     * to Terminated via the existing catch block in boot().
   328→     *
   329→     * This test uses reflection to lower the timeout threshold to 0.001s so
   330→     * the test runs fast — no point sleeping 6s in the test suite. The
   331→     * bootstrapper sleeps 0.02s which is > 0.001s threshold → throws.
   332→     *
   333→     * Verifies:
   334→     *   1. BootstrapperTimeoutExceeded is thrown when bootstrap() exceeds the
   335→     *      per-bootstrapper wall-clock budget.
   336→     *   2. The Kernel transitions to Terminated (via the existing catch block).
   337→     *   3. The exception message names the offending bootstrapper class and
   338→     *      reports both the budget and elapsed time (provenance for debugging).
   339→     */
   340→    public function testBootstrapperTimeoutExceededThrows(): void
   341→    {
   342→        $slowBootstrapper = new class implements BootstrapperInterface {
   343→            public function bootstrap(KernelInterface $kernel): void
   344→            {
   345→                // Sleep 0.02s — exceeds the 0.001s threshold we'll set via
   346→                // reflection below. In production, the threshold is 5.0s
   347→                // (Kernel::BOOTSTRAPPER_TIMEOUT_SECONDS) so this sleep would
   348→                // be well under budget; the test artificially lowers it.
   349→                \usleep(20_000);  // 20ms = 0.02s
   350→            }
   351→        };
   352→
   353→        $kernel = TestKernelFactory::create($slowBootstrapper);
   354→
   355→        // Lower the per-bootstrapper timeout to 0.001s for this test so the
   356→        // 0.02s sleep triggers the timeout. The constant
   357→        // BOOTSTRAPPER_TIMEOUT_SECONDS (5.0s) is the production hard ceiling
   358→        // and remains unchanged — we only override the instance property.
   359→        $timeout = new \ReflectionProperty(
   360→            \SovereignStack\Core\Kernel\Kernel::class,
   361→            'bootstrapperTimeoutSeconds',
   362→        );
   363→        $timeout->setValue($kernel, 0.001);
   364→
   365→        try {
   366→            $kernel->boot();
   367→            self::fail('Expected KernelException::bootstrapperTimeoutExceeded to be thrown');
   368→        } catch (KernelException $e) {
   369→            self::assertStringContainsString(
   370→                'exceeded the per-bootstrapper wall-clock budget',
   371→                $e->getMessage(),
   372→            );
   373→            self::assertStringContainsString('0.00s budget', $e->getMessage());
   374→            self::assertStringContainsString('0.02s elapsed', $e->getMessage());
   375→        }
   376→
   377→        // Per doctrine §4.5.3: Kernel MUST transition to Terminated via the
   378→        // existing catch block in boot(). The boot graph MUST be released.
   379→        self::assertSame(KernelState::Terminated, $kernel->getState());
   380→    }
