    40→async function generateContentWithFailover(client, options) {
    41→  const models = ['gemini-2.5-flash', 'gemini-2.0-flash', 'gemini-1.5-flash', 'gemini-1.5-flash-8b'];
    42→  let lastErr = null;
    43→  for (const model of models) {
    44→    try {
    45→      return await client.models.generateContent({
    46→        ...options,
    47→        model
    48→      });
    49→    } catch (err) {
    50→      if (err.status !== 404) {
    51→        console.warn(`Model ${model} failed with ${err.status || err.message}, trying next candidate...`);
    52→      }
    53→      lastErr = err;
    54→    }
    55→  }
    56→  throw lastErr;
    57→}

   553→// Local review fallback
   554→function localReviewFallback(text) {
   555→  const suggestions = [];
   556→  if (!text || text.length < 5) return suggestions;
   557→
   558→  // 1. Repeated duplicate words
   559→  const duplicateRegex = /\b([a-zA-Z]{2,})\s+\1\b/gi;
   560→  let match;
   561→  while ((match = duplicateRegex.exec(text)) !== null && suggestions.length < 5) {
   562→    suggestions.push({
   563→      original: match[0],
   564→      suggestion: match[1],
   565→      type: 'Grammar',
   566→      explanation: `Repeated word '${match[1]}' found consecutively.`
   567→    });
   568→  }
   569→
   570→  // 2. Common spelling mistakes
   571→  const commonTypos = {
   572→    teh: 'the',
   573→    recieve: 'receive',
   574→    seperate: 'separate',
   575→    definately: 'definitely',
   576→    occured: 'occurred',
   577→    untill: 'until',
   578→    truely: 'truly',
   579→    wierd: 'weird',
   580→    accomodate: 'accommodate',
   581→    goverment: 'government'
   582→  };
   583→
   584→  for (const [typo, fix] of Object.entries(commonTypos)) {
   585→    if (suggestions.length >= 8) break;
   586→    const typoRegex = new RegExp(`\\b${typo}\\b`, 'gi');
   587→    if (typoRegex.test(text)) {
   588→      suggestions.push({
   589→        original: typo,
   590→        suggestion: fix,
   591→        type: 'Spelling',
   592→        explanation: `Spelling correction: change '${typo}' to '${fix}'.`
   593→      });
   594→    }
   595→  }
   596→
   597→  // 3. Wordy filler expressions
   598→  const fillerPhrases = {
   599→    'in order to': 'to',
   600→    'at the present time': 'now',
   601→    'due to the fact that': 'because',
   602→    'in the event that': 'if',
   603→    'for the purpose of': 'to'
   604→  };
   605→
   606→  for (const [filler, concise] of Object.entries(fillerPhrases)) {
   607→    if (suggestions.length >= 10) break;
   608→    const regex = new RegExp(`\\b${filler}\\b`, 'gi');
   609→    if (regex.test(text)) {
   610→      suggestions.push({
   611→        original: filler,
   612→        suggestion: concise,
   613→        type: 'Style',
   614→        explanation: `Conciseness: simplify '${filler}' to '${concise}'.`
   615→      });
   616→    }
   617→  }
   618→
   619→  return suggestions;
   620→}

   622→// Local transform fallback
   623→function localTransformFallback(mode, input) {
   624→  const text = (input || '').trim();
   625→  if (!text) return '';
   626→
   627→  switch (mode) {
   628→    case 'shorten':
   629→      return text
   630→        .replace(/\bin order to\b/gi, 'to')
   631→        .replace(/\bat the present time\b/gi, 'now')
   632→        .replace(/\bdue to the fact that\b/gi, 'because')
   633→        .replace(/\bvery\s+/gi, '')
   634→        .replace(/\breally\s+/gi, '')
   635→        .replace(/\bextremely\s+/gi, '')
   636→        .trim();
   637→
   638→    case 'expand':
   639→      return `${text} Furthermore, this subtle development underscores the underlying tension and brings greater depth to the narrative context.`;
   640→
   641→    case 'simplify':
   642→      return text
   643→        .replace(/\butilize\b/gi, 'use')
   644→        .replace(/\bcommence\b/gi, 'start')
   645→        .replace(/\bterminate\b/gi, 'end')
   646→        .replace(/\bsubsequently\b/gi, 'then')
   647→        .trim();
   648→
   649→    case 'formal':
   650→      return text
   651→        .replace(/\bcan't\b/gi, 'cannot')
   652→        .replace(/\bwon't\b/gi, 'will not')
   653→        .replace(/\bgonna\b/gi, 'going to')
   654→        .replace(/\bwanna\b/gi, 'want to')
   655→        .trim();
   656→
   657→    case 'casual':
   658→      return text
   659→        .replace(/\bcannot\b/gi, "can't")
   660→        .replace(/\bdo not\b/gi, "don't")
   661→        .replace(/\bwill not\b/gi, "won't")
   662→        .trim();
   663→
   664→    case 'active_voice':
   665→      return text;
   666→
   667→    default:
   668→      return text;
   669→  }
   670→}

  1430→// Helper for S3 URL resolution
  1431→function buildS3Url(endpoint, bucket, key, region = 'us-east-1') {
  1432→  const cleanKey = key.replace(/^\/+/, '');
  1433→  if (endpoint && endpoint.trim()) {
  1434→    let cleanEndpoint = endpoint.trim().replace(/\/+$/, '');
  1435→    if (!cleanEndpoint.startsWith('http://') && !cleanEndpoint.startsWith('https://')) {
  1436→      cleanEndpoint = 'https://' + cleanEndpoint;
  1437→    }
  1438→    // Check if endpoint already includes bucket or if path-style
  1439→    const url = new URL(cleanEndpoint);
  1440→    if (url.hostname.includes(bucket)) {
  1441→      return `${cleanEndpoint}/${cleanKey}`;
  1442→    }
  1443→    return `${cleanEndpoint}/${encodeURIComponent(bucket)}/${cleanKey}`;
  1444→  }
  1445→  // Standard AWS S3
  1446→  return `https://${encodeURIComponent(bucket)}.s3.${region}.amazonaws.com/${cleanKey}`;
  1447→}

  1449→function getS3SigningKey(key, dateStamp, regionName, serviceName) {
  1450→  const kDate = crypto.createHmac('sha256', 'AWS4' + key).update(dateStamp).digest();
  1451→  const kRegion = crypto.createHmac('sha256', kDate).update(regionName).digest();
  1452→  const kService = crypto.createHmac('sha256', kRegion).update(serviceName).digest();
  1453→  return crypto.createHmac('sha256', kService).update('aws4_request').digest();
  1454→}

  1509→      const headers = {
  1510→        'Authorization': `Bearer ${token.trim()}`,
  1511→        'User-Agent': 'Eloqui-Backup-Engine',
  1512→        'Accept': 'application/vnd.github.v3+json'
  1513→      };
  1514→
  1515→      // Check if file already exists to obtain its SHA (for update)
  1516→      let sha = undefined;
  1517→      const checkRes = await fetch(`${apiUrl}?ref=${encodeURIComponent(branch.trim())}`, { headers });
  1518→      if (checkRes.ok) {
  1519→        const fileData = await checkRes.json();
  1520→        sha = fileData.sha;
  1521→      }
  1522→
  1523→      // Convert content to base64 if not already
  1524→      let base64Content = '';
  1525→      if (typeof content === 'string') {
  1526→        // If content is already base64 (e.g. for binary ZIP), use as is or buffer
  1527→        if (req.body.isBase64) {
  1528→          base64Content = content;
  1529→        } else {
  1530→          base64Content = Buffer.from(content, 'utf8').toString('base64');
  1531→        }
  1532→      } else {
  1533→        base64Content = Buffer.from(JSON.stringify(content, null, 2), 'utf8').toString('base64');
  1534→      }
  1535→
  1536→      const putBody = {
  1537→        message: commitMessage || `Automated Eloqui backup: ${filename} [${new Date().toISOString()}]`,
  1538→        content: base64Content,
  1539→        branch: branch.trim()
  1540→      };
  1541→      if (sha) {
  1542→        putBody.sha = sha;
  1543→      }
  1544→
  1545→      const putRes = await fetch(apiUrl, {
  1546→        method: 'PUT',
  1547→        headers: {
  1548→          ...headers,
  1549→          'Content-Type': 'application/json'
  1550→        },
  1551→        body: JSON.stringify(putBody)
  1552→      });
  1553→
  1554→      if (!putRes.ok) {
  1555→        const errText = await putRes.text();
  1556→        return res.status(putRes.status).json({ ok: false, error: `GitHub Commit error: ${errText}` });
  1557→      }
  1558→
  1559→      const commitData = await putRes.json();
  1560→      return res.json({
  1561→        ok: true,
  1562→        message: `Committed ${filename} to ${owner}/${repo} on branch ${branch}`,
  1563→        commitSha: commitData.commit?.sha,
  1564→        htmlUrl: commitData.content?.html_url || `https://github.com/${owner}/${repo}/blob/${branch}/${filePath}`,
  1565→        filePath
  1566→      });
  1567→    }
  1568→  } catch (err) {
  1569→    console.error('GitHub backup error:', err);
  1570→    return res.status(500).json({ ok: false, error: err.message });
  1571→  }

  1574→// 3. S3 / S3-Compatible Remote Upload
  1575→apiRouter.post('/backup/remote/s3', async (req, res) => {
  1576→  try {
  1577→    const { endpoint, region = 'us-east-1', bucket, accessKeyId, secretAccessKey, prefix = 'eloqui-backups', filename, content, isBase64, contentType = 'application/json' } = req.body;
  1578→    if (!bucket || !accessKeyId || !secretAccessKey || !filename) {
  1579→      return res.status(400).json({ ok: false, error: 'Missing required S3 credentials, bucket, or filename' });
  1580→    }
  1581→
  1582→    const cleanPrefix = prefix ? prefix.replace(/^\/+|\/+$/g, '') : '';
  1583→    const key = cleanPrefix ? `${cleanPrefix}/${filename}` : filename;
  1584→
  1585→    // Buffer conversion
  1586→    let buffer;
  1587→    if (isBase64) {
  1588→      buffer = Buffer.from(content, 'base64');
  1589→    } else if (typeof content === 'string') {
  1590→      buffer = Buffer.from(content, 'utf8');
  1591→    } else {
  1592→      buffer = Buffer.from(JSON.stringify(content, null, 2), 'utf8');
  1593→    }
  1594→
  1595→    const s3Url = buildS3Url(endpoint, bucket, key, region);
  1596→    const urlObj = new URL(s3Url);
  1597→    const amzDate = new Date().toISOString().replace(/[:-]|\.\d{3}/g, '');
  1598→    const dateStamp = amzDate.substring(0, 8);
  1599→    const payloadHash = crypto.createHash('sha256').update(buffer).digest('hex');
  1600→
  1601→    const canonicalHeaders = `content-length:${buffer.length}\ncontent-type:${contentType}\nhost:${urlObj.host}\nx-amz-content-sha256:${payloadHash}\nx-amz-date:${amzDate}\n`;
  1602→    const signedHeaders = 'content-length;content-type;host;x-amz-content-sha256;x-amz-date';
  1603→    const canonicalRequest = `PUT\n${urlObj.pathname}\n\n${canonicalHeaders}\n${signedHeaders}\n${payloadHash}`;
  1604→
  1605→    const credentialScope = `${dateStamp}/${region}/s3/aws4_request`;
  1606→    const stringToSign = `AWS4-HMAC-SHA256\n${amzDate}\n${credentialScope}\n${crypto.createHash('sha256').update(canonicalRequest).digest('hex')}`;
  1607→    const signingKey = getS3SigningKey(secretAccessKey.trim(), dateStamp, region, 's3');
  1608→    const signature = crypto.createHmac('sha256', signingKey).update(stringToSign).digest('hex');
  1609→
  1610→    const authHeader = `AWS4-HMAC-SHA256 Credential=${accessKeyId.trim()}/${credentialScope}, SignedHeaders=${signedHeaders}, Signature=${signature}`;
  1611→
  1612→    const s3Res = await fetch(s3Url, {
  1613→      method: 'PUT',
  1614→      headers: {
  1615→        'Host': urlObj.host,
  1616→        'Content-Type': contentType,
  1617→        'Content-Length': buffer.length.toString(),
  1618→        'x-amz-content-sha256': payloadHash,
  1619→        'x-amz-date': amzDate,
  1620→        'Authorization': authHeader
  1621→      },
  1622→      body: buffer
  1623→    });
  1624→
  1625→    if (s3Res.ok || s3Res.status === 200 || s3Res.status === 201) {
  1626→      return res.json({
  1627→        ok: true,
  1628→        message: `Successfully uploaded ${filename} to ${bucket}/${key}`,
  1629→        key,
  1630→        bucket,
  1631→        etag: s3Res.headers.get('etag'),
  1632→        location: s3Url
  1633→      });
  1634→    } else {
  1635→      const errText = await s3Res.text();
  1636→      return res.status(s3Res.status).json({ ok: false, error: `S3 upload failed (${s3Res.status}): ${errText}` });
  1637→    }
  1638→  } catch (err) {
  1639→    console.error('S3 backup error:', err);
  1640→    return res.status(500).json({ ok: false, error: err.message });
  1641→  }
  1642→});

  1722→    const whRes = await fetch(endpointUrl.trim(), {
