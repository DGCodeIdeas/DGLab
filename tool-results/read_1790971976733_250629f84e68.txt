    56→    56→    /**
    57→    57→     * Test 1: Two concurrent Fibers MUST observe independent pulse-scoped state.
    58→    58→     *
    59→    59→     * Per SPEC §43: "Request A context ≠ Request B context"
    60→    60→     */
    61→    61→    public function testConcurrentFibersObserveIndependentPulseState(): void
    62→    62→    {
    63→    63→        $container = $this->createContainerWithPulseRequestContext();
    64→    64→
    65→    65→        // Each Fiber registers its OWN RequestContext via pulse(), then verifies
    66→    66→        // it sees its own (not the other Fiber's) state after a yield point.
    67→    67→        $contextA_viewOfSelf = null;
    68→    68→        $contextB_viewOfSelf = null;
    69→    69→        $contextA_viewOfB = null; // A should NOT see B's context
    70→    70→        $contextB_viewOfA = null; // B should NOT see A's context
    71→    71→
    72→    72→        $fiberA = new Fiber(function () use (
    73→    73→            $container, &$contextA_viewOfSelf, &$contextA_viewOfB
    74→    74→        ): void {
    75→    75→            $contextA = new RequestContext(
    76→    76→                requestId: self::REQUEST_ID_A,
    77→    77→                traceId: self::TRACE_ID_A,
    78→    78→                tenantId: self::TENANT_A,
    79→    79→                startedAt: new DateTimeImmutable(),
    80→    80→            );
    81→    81→            $container->pulse(RequestContext::class, $contextA);
    82→    82→
    83→    83→            // Suspend — let Fiber B run and set its own state
    84→    84→            Fiber::suspend();
    85→    85→
    86→    86→            // After resumption: A MUST still see A's context (B should not have
    87→    87→            // overwritten A's pulse-scoped state).
    88→    88→            $contextA_viewOfSelf = $container->make(RequestContext::class);
    89→    89→
    90→    90→            // Also try to read what B set — this SHOULD be A's own context, not B's.
    91→    91→            // (The Container's pulse() WeakMap<Fiber, ...> keys on the current
    92→    92→            // Fiber, so reading RequestContext::class from A's scope returns A's
    93→    93→            // instance, not B's.)
    94→    94→            $contextA_viewOfB = $container->make(RequestContext::class);
    95→    95→        });
    96→    96→
    97→    97→        $fiberB = new Fiber(function () use (
    98→    98→            $container, &$contextB_viewOfSelf, &$contextB_viewOfA
    99→    99→        ): void {
   100→   100→            $contextB = new RequestContext(
   101→   101→                requestId: self::REQUEST_ID_B,
   102→   102→                traceId: self::TRACE_ID_B,
   103→   103→                tenantId: self::TENANT_B,
   104→   104→                startedAt: new DateTimeImmutable(),
   105→   105→            );
   106→   106→            $container->pulse(RequestContext::class, $contextB);
   107→   107→
   108→   108→            // Suspend — let A resume and check its state
   109→   109→            Fiber::suspend();
   110→   110→
   111→   111→            $contextB_viewOfSelf = $container->make(RequestContext::class);
   112→   112→            $contextB_viewOfA = $container->make(RequestContext::class);
   113→   113→        });
   114→   114→
   115→   115→        // Run the Fibers interleaved
   116→   116→        $fiberA->start(); // A registers its context, suspends
   117→   117→        $fiberB->start(); // B registers its context, suspends
   118→   118→        $fiberA->resume(); // A resumes — checks its state (should still see A)
   119→   119→        $fiberB->resume(); // B resumes — checks its state (should still see B)
   120→   120→
   121→   121→        // Assertions: each Fiber MUST see its OWN context, never the other's
   122→   122→        self::assertNotNull($contextA_viewOfSelf, 'Fiber A must have observed its own context');
   123→   123→        self::assertNotNull($contextB_viewOfSelf, 'Fiber B must have observed its own context');
   124→   124→
   125→   125→        self::assertSame(self::REQUEST_ID_A, $contextA_viewOfSelf->requestId,
   126→   126→            'Fiber A MUST see its own request_id, not B\'s (state leak = architecture defect)');
   127→   127→        self::assertSame(self::REQUEST_ID_B, $contextB_viewOfSelf->requestId,
   128→   128→            'Fiber B MUST see its own request_id, not A\'s (state leak = architecture defect)');
   129→   129→
   130→   130→        self::assertSame(self::TENANT_A, $contextA_viewOfSelf->tenantId,
   131→   131→            'Fiber A MUST see its own tenant_id, not B\'s (tenant leak = architecture defect)');
   132→   132→        self::assertSame(self::TENANT_B, $contextB_viewOfSelf->tenantId,
   133→   133→            'Fiber B MUST see its own tenant_id, not A\'s (tenant leak = architecture defect)');
   134→   134→
   135→   135→        self::assertSame(self::TRACE_ID_A, $contextA_viewOfSelf->traceId,
   136→   136→            'Fiber A MUST see its own trace_id, not B\'s');
   137→   137→        self::assertSame(self::TRACE_ID_B, $contextB_viewOfSelf->traceId,
   138→   138→            'Fiber B MUST see its own trace_id, not A\'s');
   139→   139→
   140→   140→        // Sanity check — "view of B" from A's scope is actually A's own context
   141→   141→        self::assertSame($contextA_viewOfSelf, $contextA_viewOfB,
   142→   142→            'A\'s scope MUST consistently return A\'s context');
   143→   143→    }

   367→   367→    /**
   368→   368→     * Test 6: Fiber termination MUST clear that Fiber's pulse scope — a
   369→   369→     * subsequent Fiber's make() MUST NOT observe the terminated Fiber's value,
   370→   370→     * EVEN IF the terminated Fiber object has not been garbage-collected yet.
   371→   371→     *
   372→   372→     * Per CORE-02 edge case #7: "Fiber termination and GC: When a Fiber
   373→   373→     * terminates, its entries in the WeakMap<Fiber, ...> are eligible for GC."
   374→   374→     * Per CORE-02 edge case #9: child Fiber does NOT inherit parent's pulse
   375→   375→     * bindings — isolation goes both directions.
   376→   376→     *
   377→   377→     * The structural guarantee is the WeakMap<Fiber, ...> keying: a new Fiber
   378→   378→     * has a distinct key, so it queries a DIFFERENT bucket from the terminated
   379→   379→     * Fiber's. The new Fiber's bucket is empty (it never called pulse()), so
   380→   380→     * make() falls through to autowire which throws NotFoundException for
   381→   381→     * RequestContext's required constructor params.
   382→   382→     *
   383→   383→     * This test is stronger than testCompletedFiberStateIsNotVisibleToNewFiber
   384→   384→     * (Test 4): it does NOT call unset($fiberA) before checking. The isolation
   385→   385→     * guarantee does NOT depend on PHP's GC — it depends on the WeakMap keying.
   386→   386→     */
   387→   387→    public function testFiberTerminationClearsPulseScope(): void
   388→   388→    {
   389→   389→        $container = $this->createContainerWithPulseRequestContext();
   390→   390→
   391→   391→        // Fiber A: pulses a value, then completes normally. NOTE: $fiberA is
   392→   392→        // NOT released here (no unset) — its WeakMap entry is technically
   393→   393→        // still alive. The point of this test is that isolation does NOT
   394→   394→        // depend on GC timing.
   395→   395→        $fiberA = new Fiber(function () use ($container): void {
   396→   396→            $contextA = new RequestContext(
   397→   397→                requestId: self::REQUEST_ID_A,
   398→   398→                traceId: self::TRACE_ID_A,
   399→   399→                tenantId: self::TENANT_A,
   400→   400→                startedAt: new DateTimeImmutable(),
   401→   401→            );
   402→   402→            $container->pulse(RequestContext::class, $contextA);
   403→   403→            // Fiber A completes — control returns to caller.
   404→   404→        });
   405→   405→        $fiberA->start();
   406→   406→
   407→   407→        // New Fiber B: must NOT see A's pulse-scoped value (B has its own
   408→   408→        // WeakMap bucket, which is empty — make() falls through to autowire
   409→   409→        // which throws NotFoundException for RequestContext's required params).
   410→   410→        $contextFromRequestB = null;
   411→   411→        $fiberB = new Fiber(function () use ($container, &$contextFromRequestB): void {
   412→   412→            try {
   413→   413→                $contextFromRequestB = $container->make(RequestContext::class);
   414→   414→            } catch (\SovereignStack\Core\Container\NotFoundException $e) {
   415→   415→                $contextFromRequestB = null;
   416→   416→            }
   417→   417→        });
   418→   418→        $fiberB->start();
   419→   419→
   420→   420→        self::assertNull($contextFromRequestB,
   421→   421→            'Fiber B MUST NOT observe Fiber A\'s pulse-scoped value, even ' .
   422→   422→            'before A is garbage-collected (per-Fiber WeakMap bucketing is ' .
   423→   423→            'the structural guarantee of pulse isolation)');
   424→   424→    }

   426→   426→    /**
   427→   427→     * Test 7: A new Fiber (created after an old Fiber set pulse state) MUST
   428→   428→     * NOT see the old Fiber's pulse state — explicit version with separate
   429→   429→     * Fiber objects and explicit assertions about identity.
   430→   430→     *
   431→   431→     * Per SPEC §43: "Mutable request state does not survive request completion"
   432→   432→     * Per CORE-02 edge case #8: "Fiber reuse: If a Fiber is reused (started
   433→   433→     * again after termination), it gets a fresh pulse scope."
   434→   434→     *
   435→   435→     * Distinct from Test 6: this test first verifies the OLD Fiber's
   436→   436→     * own-observation contract works (sanity), THEN verifies the NEW Fiber
   437→   437→     * does not see the old state.
   438→   438→     */
   439→   439→    public function testNewFiberDoesNotSeeOldFiberPulseState(): void
   440→   440→    {
   441→   441→        $container = $this->createContainerWithPulseRequestContext();
   442→   442→
   443→   443→        // Fiber A: pulses a value, observes its own value, completes.
   444→   444→        $contextFromA = null;
   445→   445→        $fiberA = new Fiber(function () use ($container, &$contextFromA): void {
   446→   446→            $contextA = new RequestContext(

   467→   467→        $contextFromB = null;

   472→   472→                $contextFromB = null;

   474→   474→        });
   475→   475→        $fiberB->start();
   476→   476→
   477→   477→        self::assertNull($contextFromB,
   478→   478→            'New Fiber B MUST NOT see old Fiber A\'s pulse-scoped state — ' .
   479→   479→            'WeakMap<Fiber, ...> bucketing ensures per-Pulse isolation by ' .
   480→   480→            'construction, not by convention');

   529→   529→    /**
   530→   530→     * Test 9: A nested child Fiber's pulse() MUST NOT affect the parent
   531→   531→     * Fiber's pulse scope — isolation is bidirectional.
   532→   532→     *
   533→   533→     * Per CORE-02 edge case #9: "Child Fiber does NOT inherit: parent's
   534→   534→     * pulse bindings, parent's pulse instances, parent's resolution stack."
   535→   535→     *
   536→   536→     * Symmetrically, the parent does NOT inherit the child's pulse bindings
   537→   537→     * either — pulse scope is strictly per-Fiber. A child's pulse() MUST NOT
   538→   538→     * bleed into the parent's scope (or vice versa).
   539→   539→     */
   540→   540→    public function testNestedFiberHasIndependentPulseScope(): void
   541→   541→    {
   542→   542→        $container = $this->createContainerWithPulseRequestContext();
   543→   543→
   544→   544→        $contextFromParentBefore = null;
   545→   545→        $contextFromChild = null;
   546→   546→        $contextFromParentAfter = null;
   547→   547→
   548→   548→        $fiber = new Fiber(function () use (
   549→   549→            $container,
   550→   550→            &$contextFromParentBefore,
   551→   551→            &$contextFromChild,
   552→   552→            &$contextFromParentAfter
   553→   553→        ): void {
   554→   554→            // Parent pulses its own context.
   555→   555→            $parentContext = new RequestContext(
   556→   556→                requestId: self::REQUEST_ID_A,
   557→   557→                traceId: self::TRACE_ID_A,
   558→   558→                tenantId: self::TENANT_A,
   559→   559→                startedAt: new DateTimeImmutable(),

   561→   561→            $container->pulse(RequestContext::class, $parentContext);
   562→   562→            $contextFromParentBefore = $container->make(RequestContext::class);
   563→   563→
   564→   564→            // Child Fiber pulses a DIFFERENT context — this MUST NOT bleed
   565→   565→            // into the parent's pulse scope.
   566→   566→            $child = new Fiber(function () use ($container, &$contextFromChild): void {
   567→   567→                $childContext = new RequestContext(
   568→   568→                    requestId: self::REQUEST_ID_B,
   569→   569→                    traceId: self::TRACE_ID_B,
   570→   570→                    tenantId: self::TENANT_B,
   571→   571→                    startedAt: new DateTimeImmutable(),
   572→   572→                );
   573→   573→                $container->pulse(RequestContext::class, $childContext);
   574→   574→                $contextFromChild = $container->make(RequestContext::class);
   575→   575→            });
   576→   576→            $child->start();
   577→   577→
   578→   578→            // After the child returned: parent's pulse scope MUST still hold
   579→   579→            // the parent's context, not the child's. Isolation is bidirectional.

   590→   590→            'Parent MUST STILL see its own context after the child returned — ' .

   592→   592→            'isolation: parent does NOT inherit child\'s pulse bindings either)');
   593→   593→    }
