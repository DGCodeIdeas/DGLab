    15→## 1. The Deployment Principle
    16→
    17→The DGLab wheel is **never modified in place**. It is rebuilt, tested, and replaced as a complete unit. Every deployment is an atomic swap — the old wheel stops, the new wheel starts. There is no "patching" or "hotfixing" a running instance.
    18→
    19→```
    20→Immutable Deployment:
    21→
    22→   Build          Test           Deploy           Verify
    23→    │              │               │                │
    24→    ▼              ▼               ▼                ▼
    25→┌────────┐    ┌────────┐     ┌────────┐      ┌────────┐
    26→│  Code  │───►│  CI    │────►│  Blue  │─────►│ Health │
    27→│ Change │    │ Pipeline│     │ /Green │      │ Checks │
    28→└────────┘    └────────┘     └────────┘      └────────┘
    29→                                  │
    30→                                  ▼
    31→                            ┌────────┐
    32→                            │  Roll  │
    33→                            │  Back  │ (if health fails)
    34→                            └────────┘
    35→```
    36→
    37→**Rule:** If a deployment fails health checks, it is automatically rolled back. No human decision is required for rollback.

    93→
    94→```hcl
    95→# deploy/terraform/modules/vpc/main.tf
    96→
    97→resource "aws_vpc" "main" {
    98→  cidr_block           = "10.0.0.0/16"
    99→  enable_dns_hostnames = true
   100→  enable_dns_support   = true
   101→
   102→  tags = {
   103→    Name        = "sovereign-${var.environment}"
   104→    Environment = var.environment
   105→    ManagedBy   = "terraform"
   106→  }
   107→}
   108→
   109→# Public subnets (Outer Rim — CDN, LB, BRIDGE-01)
   110→resource "aws_subnet" "public" {
   111→  count                   = 3
   112→  vpc_id                  = aws_vpc.main.id
   113→  cidr_block              = "10.0.${count.index + 1}.0/24"
   114→  availability_zone       = data.aws_availability_zones.available.names[count.index]
   115→  map_public_ip_on_launch = true
   116→
   117→  tags = {
   118→    Name = "sovereign-${var.environment}-public-${count.index + 1}"
   119→    Tier = "public"
   120→  }
   121→}
   122→
   123→# Private subnets (Hub + Inner Spokes)

   517→## 4. CI/CD Pipeline
   518→
   519→### 4.1 GitHub Actions Workflow
   520→
   521→```yaml
   522→# .github/workflows/deploy.yml
   523→name: Deploy
   524→
   525→on:
   526→  push:
   527→    branches: [main]
   528→  pull_request:
   529→    branches: [main]
   530→
   531→env:
   532→  REGISTRY: ghcr.io
   533→  IMAGE_NAME: ${{ github.repository }}
   534→
   535→jobs:
   536→  build:
   537→    runs-on: ubuntu-latest
   538→    strategy:
   539→      matrix:
   540→        component: [core, hub, bridge, spokes-internal, spokes-external]
   541→    steps:
   542→      - uses: actions/checkout@v4
   543→
   544→      - name: Set up Docker Buildx
   545→        uses: docker/setup-buildx-action@v3
   546→
   547→      - name: Log in to Container Registry
   548→        uses: docker/login-action@v3
   549→        with:
   550→          registry: ${{ env.REGISTRY }}
   551→          username: ${{ github.actor }}
   552→          password: ${{ secrets.GITHUB_TOKEN }}
   553→
   554→      - name: Extract metadata
   555→        id: meta
   556→        uses: docker/metadata-action@v5
   557→        with:
   558→          images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}/${{ matrix.component }}
   559→          tags: |
   560→            type=sha,prefix={{branch}}-
   561→            type=semver,pattern={{version}}
   562→            type=raw,value=latest,enable={{is_default_branch}}
   563→
   564→      - name: Build and push
   565→        uses: docker/build-push-action@v5
   566→        with:
   567→          context: .
   568→          file: ./docker/${{ matrix.component }}.Dockerfile
   569→          push: ${{ github.event_name != 'pull_request' }}
   570→          tags: ${{ steps.meta.outputs.tags }}
   571→          labels: ${{ steps.meta.outputs.labels }}
   572→          cache-from: type=gha
   573→          cache-to: type=gha,mode=max
   574→
   575→  test:
   576→    needs: build
   577→    runs-on: ubuntu-latest
   578→    steps:
   579→      - uses: actions/checkout@v4
   580→      - uses: shivammathur/setup-php@v2
   581→        with: { php-version: '8.3' }
   582→      - run: composer install
   583→      - run: vendor/bin/phpunit --testsuite=unit
   584→      - run: vendor/bin/phpunit --testsuite=integration
   585→      - run: vendor/bin/phpstan analyse
   586→      - run: composer audit
   587→
   588→  deploy-dev:
   589→    needs: [build, test]
   590→    if: github.ref == 'refs/heads/main'
   591→    runs-on: ubuntu-latest
   592→    environment: development
   593→    steps:
   594→      - uses: actions/checkout@v4
   595→
   596→      - name: Configure AWS credentials
   597→        uses: aws-actions/configure-aws-credentials@v4
   598→        with:
   599→          role-to-assume: ${{ secrets.AWS_DEPLOY_ROLE_ARN }}
   600→          aws-region: us-east-1
   601→
   602→      - name: Update kubeconfig
   603→        run: aws eks update-kubeconfig --name sovereign-dev
   604→
   605→      - name: Deploy to development
   606→        run: |
   607→          kubectl set image deployment/vanguard             vanguard=ghcr.io/${{ github.repository }}/bridge:${{ github.sha }}             -n sovereign-bridge
   608→          kubectl rollout status deployment/vanguard -n sovereign-bridge
   609→
   610→      - name: Run smoke tests
   611→        run: |
   612→          curl -f https://dev.sovereign.example/health/ready
   613→          curl -f https://dev.sovereign.example/health/live
   614→
   615→  deploy-staging:
   616→    needs: deploy-dev
   617→    runs-on: ubuntu-latest
   618→    environment: staging
   619→    steps:
   620→      - uses: actions/checkout@v4
   621→      - name: Deploy to staging
   622→        run: |
   623→          # Similar to dev but with staging EKS cluster
   624→          echo "Deploying to staging..."
   625→
   626→      - name: Run integration tests against staging
   627→        run: |
   628→          BASE_URL=https://staging.sovereign.example             vendor/bin/phpunit --testsuite=e2e
   629→
   630→  deploy-production:
   631→    needs: deploy-staging
   632→    runs-on: ubuntu-latest
   633→    environment: production
   634→    steps:
   635→      - uses: actions/checkout@v4
   636→      - name: Deploy to production (canary)
   637→        run: |
   638→          # ArgoCD handles canary deployment
   639→          # This step just triggers the sync
   640→          argocd app sync sovereign-production
   641→
   642→      - name: Verify canary health
   643→        run: |
   644→          sleep 60
   645→          curl -f https://sovereign.example/health/ready
   646→```

   927→
   928→### 8.1 Runbook: Complete Region Failure
   929→
   930→```
   931→SCENARIO: AWS us-east-1 complete outage
   932→
   933→STEP 1: Activate standby region
   934→    $ aws route53 change-resource-record-sets         --hosted-zone-id Z123456789         --change-batch file://failover-to-standby.json
   935→
   936→STEP 2: Verify standby EKS cluster
   937→    $ kubectl --context=sovereign-standby get nodes
   938→    $ kubectl --context=sovereign-standby get pods --all-namespaces
   939→
   940→STEP 3: Promote RDS read replica to primary
   941→    $ aws rds promote-read-replica         --db-instance-identifier sovereign-standby-primary
   942→
   943→STEP 4: Verify data consistency
   944→    $ php bin/console audit:verify-chain
   945→    $ php bin/console health:check --all
   946→
   947→STEP 5: Notify stakeholders
   948→    $ php bin/console notify:send         --channel=ops         --message="Failover to standby complete. RTO: XX minutes."
   949→
   950→STEP 6: Post-incident review
   951→    - Document actual RTO/RPO
   952→    - Identify gaps
   953→    - Update runbook
   954→```
