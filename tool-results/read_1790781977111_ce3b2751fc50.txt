    17→export function htmlToBlocks(html: string): Block[] {
    18→  if (!html || !html.trim()) {
    19→    return [{ id: crypto.randomUUID(), type: 'paragraph', content: '' }];
    20→  }
    21→  try {
    22→    const parser = new DOMParser();
    23→    const doc = parser.parseFromString(html, 'text/html');
    24→    const blocks: Block[] = [];
    25→    const children = Array.from(doc.body.children);
    26→
    27→    if (children.length === 0) {
    28→      const text = doc.body.textContent || '';
    29→      return [{ id: crypto.randomUUID(), type: 'paragraph', content: text }];
    30→    }
    31→
    32→    for (const child of children) {
    33→      const tag = child.tagName.toLowerCase();
    34→      const text = (child.textContent || '').trim();
    35→      if (tag === 'h1') {
    36→        blocks.push({ id: crypto.randomUUID(), type: 'h1', content: text });
    37→      } else if (tag === 'h2') {
    38→        blocks.push({ id: crypto.randomUUID(), type: 'h2', content: text });
    39→      } else if (tag === 'ul') {
    40→        const lis = Array.from(child.querySelectorAll('li'));
    41→        if (lis.length > 0) {
    42→          lis.forEach(li => {
    43→            blocks.push({ id: crypto.randomUUID(), type: 'bullet', content: (li.textContent || '').trim() });
    44→          });
    45→        } else {
    46→          blocks.push({ id: crypto.randomUUID(), type: 'bullet', content: text });
    47→        }
    48→      } else if (tag === 'pre' || tag === 'code') {
    49→        blocks.push({ id: crypto.randomUUID(), type: 'code', content: text });
    50→      } else if (tag === 'blockquote') {
    51→        blocks.push({ id: crypto.randomUUID(), type: 'quote', content: text });
    52→      } else if (tag === 'hr') {
    53→        blocks.push({ id: crypto.randomUUID(), type: 'divider', content: '' });
    54→      } else {
    55→        blocks.push({ id: crypto.randomUUID(), type: 'paragraph', content: text });
    56→      }
    57→    }
    58→
    59→    return blocks.length > 0 ? blocks : [{ id: crypto.randomUUID(), type: 'paragraph', content: '' }];
    60→  } catch {
    61→    return [{ id: crypto.randomUUID(), type: 'paragraph', content: html.replace(/<[^>]*>/g, '') }];
    62→  }
    63→}

    65→export function blocksToHtml(blocks: Block[]): string {
    66→  if (!blocks || blocks.length === 0) return '<p></p>';
    67→  return blocks.map(b => {
    68→    switch (b.type) {
    69→      case 'h1': return `<h1>${b.content}</h1>`;
    70→      case 'h2': return `<h2>${b.content}</h2>`;
    71→      case 'bullet': return `<ul><li>${b.content}</li></ul>`;
    72→      case 'code': return `<pre><code>${b.content}</code></pre>`;
    73→      case 'quote': return `<blockquote>${b.content}</blockquote>`;
    74→      case 'divider': return `<hr/>`;
    75→      default: return `<p>${b.content}</p>`;
    76→    }
    77→  }).join('');
    78→}

   431→  detectSensitiveTerms(text: string): { count: number; terms: string[] } {
   432→    if (!text) return { count: 0, terms: [] };
   433→    const plainText = text.replace(/<[^>]*>/g, ' ');
   434→    const terms = this.allSensitiveTerms();
   435→    let count = 0;
   436→    const detected = new Set<string>();
   437→
   438→    for (const term of terms) {
   439→      if (!term.trim()) continue;
   440→      const escaped = term.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
   441→      const regex = new RegExp(`\\b${escaped}\\b`, 'gi');
   442→      const matches = plainText.match(regex);
   443→      if (matches && matches.length > 0) {
   444→        count += matches.length;
   445→        detected.add(term.toLowerCase());
   446→      }
   447→    }
   448→    return { count, terms: Array.from(detected) };
   449→  }

   451→  redactHtml(html: string): string {
   452→    if (!html) return '';
   453→    // Uncensored Mode: Return full raw creative text without any redactions
   454→    if (this.isUncensored()) {
   455→      return html;
   456→    }
   457→
   458→    const terms = this.allSensitiveTerms();
   459→    if (terms.length === 0) return html;
   460→
   461→    const style = this.redactionStyle();
   462→    const escapedTerms = terms
   463→      .map(t => t.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
   464→      .filter(Boolean)
   465→      .join('|');
   466→    
   467→    if (!escapedTerms) return html;
   468→    const regex = new RegExp(`\\b(${escapedTerms})\\b`, 'gi');
   469→
   470→    return html.replace(/(>|^)([^<]+)(<|$)/g, (match, prefix, text, suffix) => {
   471→      const replacedText = text.replace(regex, (word: string) => {
   472→        if (style === 'blackbar' || style === 'blackout') {
   473→          return `<span class="censored-blackbar" title="Censored term: Click to reveal" onclick="this.classList.toggle('unredacted')">${'█'.repeat(Math.max(4, word.length))}</span>`;
   474→        } else if (style === 'blur') {
   475→          return `<span class="censored-blur" title="Censored term: Click to reveal" onclick="this.classList.toggle('unredacted')">${word}</span>`;
   476→        } else if (style === 'asterisks') {
   477→          return `<span class="censored-spoiler" title="Censored term: Click to reveal" onclick="this.classList.toggle('unredacted')">${'*'.repeat(Math.max(3, word.length))}</span>`;
   478→        } else if (style === 'redact_pill') {
   479→          return `<span class="censored-pill" title="Censored term: Click to reveal" onclick="this.classList.toggle('unredacted')">[REDACTED]</span>`;
   480→        } else {
   481→          return `<span class="censored-spoiler" title="Censored term: Click to reveal" onclick="this.classList.toggle('unredacted')">${word}</span>`;
   482→        }
   483→      });
   484→      return prefix + replacedText + suffix;
   485→    });
   486→  }

   537→  addCustomBlacklistTerm(term: string) {
   538→    const clean = term.trim().toLowerCase();
   539→    if (!clean) return;
   540→    this.currentDoc.update(doc => {
   541→      if (!doc) return null;
   542→      const config = doc.censorshipConfig || { redactionStyle: 'spoiler', customBlacklist: [] };
   543→      const currentList = config.customBlacklist || [];
   544→      if (currentList.includes(clean)) return doc;

   771→  async downloadDocument(format: 'html' | 'txt' | 'markdown' | 'json' | 'epub' | 'pdf', options?: { respectRedactions?: boolean }) {
   772→     const doc = this.currentDoc();
   773→     if (!doc) return;
   774→
   775→     const shouldRedact = options?.respectRedactions !== undefined ? options.respectRedactions : !this.isUncensored();
   776→     const style = this.redactionStyle();
   777→
   778→     if (format === 'epub') {
   779→         await this.exportEpub(doc, shouldRedact);
   780→         return;
   781→     }
   782→
   783→     if (format === 'pdf') {
   784→       const { jsPDF } = await import('jspdf');
   785→       const pdf = new jsPDF({
   786→         orientation: 'portrait',
   787→         unit: 'pt',
   788→         format: 'letter'
   789→       });
   790→       const margin = 54;
   791→       const pageWidth = pdf.internal.pageSize.getWidth();
   792→       const printableWidth = pageWidth - (margin * 2);
   793→       let currentY = margin;
   794→
   795→       pdf.setFont('times', 'bold');
   796→       pdf.setFontSize(22);
   797→       const titleLines = pdf.splitTextToSize(doc.title || 'Untitled', printableWidth);
   798→       pdf.text(titleLines, margin, currentY + 16);
   799→       currentY += (titleLines.length * 26) + 20;
   800→
   801→       for (let i = 0; i < doc.chapters.length; i++) {
   802→         const ch = doc.chapters[i];
   803→         if (i > 0) {
   804→           pdf.addPage();
   805→           currentY = margin + 20;
   806→         }
   807→         pdf.setFont('times', 'bold');
   808→         pdf.setFontSize(15);
   809→         pdf.text(ch.title || `Chapter ${i + 1}`, margin, currentY);
   810→         currentY += 22;
   811→
   812→         let raw = ch.content || '';
   813→         if (shouldRedact) {
   814→           raw = this.redactHtml(raw);
   815→         }
   816→         const div = document.createElement('div');
   817→         div.innerHTML = raw.replace(/<\/p>/gi, '\n\n').replace(/<br\s*[\/]?>/gi, '\n');
   818→         const text = (div.textContent || '').trim();
   819→         
   820→         pdf.setFont('times', 'normal');
   821→         pdf.setFontSize(11);
   822→         const paragraphs = text.split('\n\n').filter(p => p.trim().length > 0);
   823→         for (const p of paragraphs) {
   824→           const lines = pdf.splitTextToSize(p, printableWidth);
   825→           for (const line of lines) {
   826→             if (currentY > pdf.internal.pageSize.getHeight() - margin - 30) {
   827→               pdf.addPage();
   828→               currentY = margin + 20;
   829→               pdf.setFont('times', 'normal');
   830→               pdf.setFontSize(11);
   831→             }
   832→             pdf.text(line, margin, currentY);
   833→             currentY += 15;
   834→           }
   835→           currentY += 6;
   836→         }
   837→       }
   838→
   839→       pdf.save(`${(doc.title || 'Untitled').replace(/[^a-z0-9]/gi, '_').toLowerCase()}.pdf`);
   840→       return;
   841→     }
   842→     
   843→     // Join all chapters for other formats
   844→     let fullContent = doc.chapters.map(c => {
   845→       const raw = c.content || '';
   846→       return shouldRedact ? this.redactHtml(raw) : raw;
   847→     }).join('\n\n<hr class="chapter-break" />\n\n');
   848→     
   849→     if (format === 'txt') {
   850→         const div = document.createElement('div');
   851→         div.innerHTML = fullContent;
   852→         let text = div.textContent || '';
   853→         if (shouldRedact) {
   854→           const terms = this.allSensitiveTerms();
   855→           if (terms.length > 0) {
   856→             const escaped = terms.map(t => t.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&')).filter(Boolean).join('|');
   857→             if (escaped) {
   858→               const regex = new RegExp(`\\b(${escaped})\\b`, 'gi');
   859→               text = text.replace(regex, (m: string) => {
   860→                 if (style === 'blackbar' || style === 'blackout') return '█'.repeat(Math.max(4, m.length));
   861→                 if (style === 'asterisks') return '*'.repeat(Math.max(3, m.length));
   862→                 return '[REDACTED]';
   863→               });
   864→             }
   865→           }
   866→         }
   867→         fullContent = text;
   868→     } else if (format === 'markdown') {
   869→        let md = fullContent
   870→           .replace(/<h1>(.*?)<\/h1>/g, '# $1\n\n')
   871→           .replace(/<h2>(.*?)<\/h2>/g, '## $1\n\n')
   872→           .replace(/<p>(.*?)<\/p>/g, '$1\n\n')
   873→           .replace(/<[^>]*>/g, '');
   874→        if (shouldRedact) {
   875→          const terms = this.allSensitiveTerms();
   876→          if (terms.length > 0) {
   877→            const escaped = terms.map(t => t.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&')).filter(Boolean).join('|');
   878→            if (escaped) {
   879→              const regex = new RegExp(`\\b(${escaped})\\b`, 'gi');
   880→              md = md.replace(regex, (m: string) => {
   881→                if (style === 'blackbar' || style === 'blackout') return '█'.repeat(Math.max(4, m.length));
   882→                if (style === 'asterisks') return '*'.repeat(Math.max(3, m.length));
   883→                return '[REDACTED]';
   884→              });
   885→            }
   886→          }
   887→        }
   888→        fullContent = md;
   889→     }
   890→
   891→     const mime = format === 'json' ? 'application/json' : (format === 'txt' ? 'text/plain' : 'text/html');
   892→     const ext = format === 'markdown' ? 'md' : format;
   893→     
   894→     const finalData = format === 'json' ? JSON.stringify(doc, null, 2) : fullContent;
   895→
   896→     const blob = new Blob([finalData], { type: mime });
   897→     const url = URL.createObjectURL(blob);
   898→     const a = document.createElement('a');
   899→     a.href = url;
   900→     a.download = `${(doc.title || 'Untitled').replace(/[^a-z0-9]/gi, '_').toLowerCase()}.${ext}`;
   901→     a.click();
   902→     URL.revokeObjectURL(url);
   903→  }

   992→  /**
   993→   * Restores vault documents from a backup payload
   994→   */
   995→  async restoreVault(restoredDocs: Document[], mode: 'merge' | 'replace'): Promise<void> {
   996→    if (!restoredDocs || restoredDocs.length === 0) return;
   997→
   998→    if (mode === 'replace') {
   999→      // Delete existing documents
  1000→      const current = this.documents();
  1001→      for (const d of current) {
  1002→        await this.storage.deleteDocument(d.id);
  1003→      }
  1004→      // Save all restored documents
  1005→      for (const d of restoredDocs) {
  1006→        await this.storage.saveDocument(d);
  1007→      }
  1008→      this.documents.set(restoredDocs);
  1009→      if (restoredDocs.length > 0) {
  1010→        this.selectDocument(restoredDocs[0].id);
  1011→      }
  1012→    } else {
  1013→      // Merge mode
  1014→      const existing = [...this.documents()];
  1015→      const existingMap = new Map(existing.map(d => [d.id, d]));
  1016→
  1017→      for (const d of restoredDocs) {
  1018→        if (existingMap.has(d.id)) {
  1019→          // If already exists, generate a new ID or replace if newer
  1020→          const existingDoc = existingMap.get(d.id)!;
  1021→          if (d.lastModified > existingDoc.lastModified) {
  1022→            existingMap.set(d.id, d);
  1023→            await this.storage.saveDocument(d);
  1024→          } else {
  1025→            // Keep local newer, or add as copy if titles differ
  1026→            const copyDoc: Document = {
  1027→              ...d,
  1028→              id: crypto.randomUUID(),
  1029→              title: `${d.title} (Restored Copy)`
  1030→            };
  1031→            existingMap.set(copyDoc.id, copyDoc);
  1032→            await this.storage.saveDocument(copyDoc);
  1033→          }
  1034→        } else {
  1035→          existingMap.set(d.id, d);
  1036→          await this.storage.saveDocument(d);
  1037→        }
  1038→      }
  1039→
  1040→      const merged = Array.from(existingMap.values());
  1041→      this.documents.set(merged);
  1042→      if (restoredDocs.length > 0) {
  1043→        this.selectDocument(restoredDocs[0].id);
  1044→      }
  1045→    }
  1046→  }
  1047→}
  1048→
